Inline 6510 assembly

Use source operands, follow the preservation wrapper, and call a ROM routine or terminal vector.

Start with a block-local loop

A statement-level asm { ... } block uses 64tass-style instruction syntax. ZeroPage extracts the block before Zig validates the surrounding source. This loop uses X, modifies the mutable byte color, and writes to the VIC-II border register.

asm-border.zp64
const c64 = @import("c64");

export fn main() void {
    c64.console.clear();
    var color: u8 = 2;
    asm {
        ldx #3
    brighten:
        inc @color
        dex
        bne brighten
        lda @color
        sta $d020
    }
    c64.console.print("BORDER {x}\n", .{color});
}

Build and run this example

Expect BORDER 05 and a green border. brighten: belongs to this block. Another block may reuse the label. Branches target labels inside their own block and have a displacement of −128 to +127 bytes from the next instruction. jmp label also stays inside the block.

Parameters, locals, enums, and flattened record fields

@delta names a byte parameter cell. @result names a mutable byte local. Use @actor_pos_x for the flattened byte field actor.pos.x. This works for a by-value record parameter and a local record. The enum field uses the byte tag from @mode. The function returns through normal ZeroPage source.

asm-record.zp64
const c64 = @import("c64");

const Mode = enum(u8) { normal = 1, bright = 2 };
const Position = struct { x: u8, y: u8 };
const Actor = struct { pos: Position, mode: Mode };

fn adjustedX(actor: Actor, delta: u8) u8 {
    var result: u8 = 0;
    asm {
        lda @actor_pos_x
        clc
        adc @delta
        sta @result
    }
    return result;
}

export fn main() void {
    c64.console.clear();
    var actor = Actor{ .pos = Position{ .x = 3, .y = 4 }, .mode = .normal };
    var mode: Mode = .bright;
    asm {
        inc @actor_pos_x
        lda @mode
        sta @actor_mode
    }
    const x = adjustedX(actor, 1);
    c64.console.print("X {x} MODE {x}\n", .{ x, actor.mode });
}

Build and run this example

Expect X 05 MODE 02. Use u8, i8, bool, or byte enums and byte fields. Keep Boolean and enum values valid. Treat parameter cells as the callee's copies. Plain scalar const bindings and word, pointer, array, or whole-record operands are outside this documented interface. Use ZeroPage code to split wider inputs or prepare pointer accesses. #@delta is invalid because a variable cell is not an immediate.

String addresses, indexed data, and length cells

@source.data,y reads a byte using Y. @msg.data,y writes the mutable buffer. #@source.len is an immediate length, while @msg.len is a runtime cell. A buffer's capacity is fixed. An assembly copy must respect it and update the length.

16_inline_asm_strings.zp64
const c64 = @import("c64");

const source = "asm string ok\n";

export fn main() void {
    c64.console.clear();
    c64.console.print("inline asm strings\n", .{});

    var msg = c64.string.buffer(32);
    asm {
        ldy #0
copy:
        lda @source.data,y
        sta @msg.data,y
        iny
        cpy #@source.len
        bne copy
        lda #@source.len
        sta @msg.len
        lda #33
        sta @msg.data+12
    }

    c64.console.puts(msg);
    c64.console.print("asm string done\n", .{});
    c64.vic.border = 5;
}
Load a data address into X/Y · excerpt
ldx #<@source.data
ldy #>@source.data

The same low/high address forms work for a mutable buffer. An offset such as @msg.data+12 remains assembler arithmetic. Constant data is read-only. #@msg.len and bare string operands are rejected.

To read indirectly, reserve a zero-page pair, initialize it with the low and high address bytes, then load through ($30),y. This example saves and restores the borrowed BASIC bytes itself. The register wrapper preserves CPU state, and the compiler's zero-page cleanup covers its own claims.

asm-indirect.zp64
const c64 = @import("c64");
const text = "A";

memory {
    zero_page {
        profile = .kernal;
        reserve at 0x30 size 2;
        restore_basic = true;
        interrupts = .system;
    }
}

export fn main() void {
    c64.console.clear();
    var glyph: u8 = 0;
    var saved_lo: u8 = 0;
    var saved_hi: u8 = 0;
    asm {
        lda $30
        sta @saved_lo
        lda $31
        sta @saved_hi
        lda #<@text.data
        sta $30
        lda #>@text.data
        sta $31
        ldy #0
        lda ($30),y
        sta @glyph
        lda @saved_lo
        sta $30
        lda @saved_hi
        sta $31
    }
    c64.console.print("GLYPH {x}\n", .{glyph});
}

What the compiler saves and restores

The wrapper combines clobbers across the whole block. The following table counts wrapper bytes on the hardware stack. A subroutine's return address, nested calls, and interrupts consume additional stack space.

sta $d020, nopNone0
inc @value, cmp #5, clcP1
lda #5, asl aP and A2
ldx #3 or ldy #3P, A, and the changed index register3

External calls, terminal jumps, and rejected forms

External jsr accepts a numeric address or an assembler-resolved symbol and must return with the hardware stack balanced. Local jsr, inline rts/rti/brk, and stack manipulation such as pha or txs are rejected. Keep directives, includes, macros, and data declarations outside the inline form. Use a static-frame function. Stack frames and DLL exports have stricter body rules.

ReadyOS terminal transfer · requires the resident shim
asm {
    jmp $c80c
}

This single-instruction absolute block transfers to a documented resident vector and bypasses normal ZeroPage return. Supply the machine state required by the vector. Prepare required memory in ZeroPage. A preceding normal block restores registers and flags at fallthrough. Indirect jumps, external symbolic jumps, and additional instructions in the terminal block are rejected.
