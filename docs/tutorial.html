Build with ZeroPage

Build the compiler, write a program, and follow the generated code from C64 RAM to a DLL loaded from disk.

Install the host tools

Download and unpack Zig 0.16.0 and install 64tass 1.60.3243. Put the directories containing the two executables on your shell's PATH. Zig builds the host compiler and provides its parser. 64tass assembles the emitted 6510 code. Install VICE for x64sc and the optional c1541 disk utility. Zig downloads · 64tass downloads · VICE downloads

git clone https://github.com/ReadyOS-C64/ZeroPageLang.git

zig build installs zp64c under zig-out/bin/. To use an assembler outside PATH, set ZEROPAGE_64TASS=/absolute/path/to/64tass. Tool overrides · Dependency reference

Build and run a PRG

The compiler starts at export fn main() void. The c64 import provides the machine-facing API. .{} is an empty argument tuple for the formatter.

hello.zp64
const c64 = @import("c64");

export fn main() void {
    c64.console.clear();
    c64.console.print("HELLO FROM ZEROPAGE\n", .{});
    c64.vic.border = 5;
}
Terminal
mkdir -p build
./zig-out/bin/zp64c compile examples/hello.zp64 -o build/hello.prg
x64sc -autostart build/hello.prg

The default image loads at $0801. A BASIC line runs SYS 2061, entering the compiled program at $080d. Expect a greeting and a green border. On hardware, put the PRG on a disk or compatible storage device, load it, and enter RUN.

Write typed 6510 code

Save the example as tutorial-colors.zp64 in the repository root. Each keypress advances to the next color. Return exits the program.

tutorial-colors.zp64
const c64 = @import("c64");

const colors = [_]u8{ 2, 5, 6 };

fn nextIndex(index: u8) u8 {
    if (index == 2) {
        return 0;
    }
    return index + 1;
}

export fn main() void {
    var index: u8 = 0;
    c64.console.clear();
    c64.vic.border = colors[index];
    c64.screen.hexAt(0, 0, index);
    var pressed = c64.keyboard.waitPetscii();
    while (pressed != c64.key.return_) {
        index = nextIndex(index);
        c64.vic.border = colors[index];
        c64.screen.hexAt(0, 0, index);
        pressed = c64.keyboard.waitPetscii();
    }
}
Terminal
./zig-out/bin/zp64c compile tutorial-colors.zp64 -o build/tutorial-colors.prg
x64sc -autostart build/tutorial-colors.prg

The fixed array has three bytes. nextIndex keeps the runtime index within bounds because generated indexing has no runtime bounds check. Use u8 for byte values and u16 or usize for words and addresses. Types, control flow, and language limits

Talk to the hardware

The border helper maps to the VIC-II register at $d020. The example below uses a volatile pointer to that register, then an assembly block to write the background register at $d021. Save it as tutorial-hardware.zp64 and compile it using the same command pattern.

tutorial-hardware.zp64
export fn main() void {
    const border: *volatile u8 = @ptrFromInt(0xd020);
    border.* = 5;
    asm {
        lda #$06
        sta $d021
    }
}

Use c64.screen for direct screen writes, c64.console for PETSCII output and formatting, and c64.kernal when you need the ROM routines. The keyboard API reports PETSCII values. Inline assembly has a documented instruction and operand subset. C64 API · Inline assembly rules

Plan the memory layout

Open the memory placement demo. It declares visible code at $2000, an arena at $9000, and a region at $e000 behind KERNAL ROM. The resident image starts from its configured entry at $1000.

95_memory_regions_placement.zp64 · placement excerpt
const c64 = @import("c64");

const hidden_values linksection("hidden") = [_]u8{ 0x21, 0x42, 0x63 };

memory {
    region direct_code: c64 at 0x2000 size 0x0100;
    region near_arena: c64 at 0x9000 size 0x0100;
    region hidden_code: c64 at 0xe000 size 0x0100 bank .ram_under_kernal;

    placement direct in direct_code;
    placement hidden in hidden_code load .startup_copy;
    arena near_heap in near_arena;

    image prg {
        load_addr = 0x0801;
        entry_addr = 0x1000;
        basic_stub = true;
        address_limit = 0xc600;
    }
}

fn directAdd(value: u8) linksection("direct") u8 {
    return value + 1;
}

fn hiddenLookup(index: u8) linksection("hidden") u8 {
    return hidden_values[index];
}

linksection("hidden") places the lookup function and table together. .startup_copy keeps their bytes in the load image and copies them to the run address before main. A resident call gateway exposes the hidden RAM and restores the previous mapping. The demo also uses near_heap.allocBytes(2), which returns an optional pointer to visible C64 RAM.

Terminal
./zig-out/bin/zp64c compile examples/small/95_memory_regions_placement.zp64 -o build/placement.prg
x64sc -autostart build/placement.prg

Expect PLACEMENT ARENA OK. Open build/placement.prg.zpmap.json and compare the load and run addresses of the hidden placement. A named pool allocates equal-sized blocks, while an arena releases its allocations together with reset(). The separate zero_page policy governs borrowed bytes and fixed reservations. Use a software stack

Ordinary functions use fixed frame storage. For recursion, reserve RAM for the software stack and mark the recursive function with callconv(.stack). This example counts down to zero and returns the green border value. An exhausted software stack traps.

32_stack_recursive_border.zp64
const c64 = @import("c64");

memory {
    region call_stack: c64 at 0xc800 size 0x0400;

    stack in call_stack;
}

fn borderFrom(value: u8) callconv(.stack) u8 {
    if (value == 0) {
        return 5;
    }
    return borderFrom(value - 1);
}

export fn main() callconv(.stack) void {
    var border: u8 = borderFrom(4);
    c64.vic.border = border;
}

Compile the file and inspect the generated assembly for the entry and frame setup. The software stack holds the supported call-frame values. The 6510 hardware stack still holds CPU return addresses. Frame layouts and ABI limits

Load a DLL from disk

Open the disk-load example. The loader uses visible RAM at $9000 for disk staging, REU storage at $030000 for the cache, and a C64 execution window at $c000. The DLL declares an ABI and an export order. Each export uses the software-stack calling convention.

51_module_disk_load_real.zp64
const c64 = @import("c64");
const ZeroPage = @import("ZeroPage");

memory {
    region call_stack: c64 at 0xc800 size 0x0400;
    region module_window: c64 at 0xc000 size 0x0800;
    region module_cache: reu at 0x030000 size 0x010000;
    region staging_ram: c64 at 0x9000 size 0x0800;

    stack in call_stack;
}

module_loader {
    window main in module_window;
    reu_cache in module_cache;
    disk_staging in staging_ram;
}

module enemy_runtime abi 1 {
    export border = borderColor;
}

fn borderColor(seed: u8) callconv(.stack) u8 {
    return seed + 2;
}

export fn main() void {
    c64.console.clear();
    c64.console.print("DISK LOAD REAL\n", .{});
    if (ZeroPage.module.loadFromDisk("enemy_runtime", "enemyrt")) {
        const maybe_border = ZeroPage.module.call("enemy_runtime", "border", .{3});
        if (maybe_border) |border| {
            c64.console.print("REAL LOAD OK\n", .{});
            c64.vic.border = border;
        } else {
            c64.console.print("REAL CALL FAIL\n", .{});
            c64.vic.border = 2;
        }
    } else {
        c64.console.print("REAL LOAD FAIL\n", .{});
        c64.vic.border = 2;
    }
}
Terminal
bash scripts/build_sample_disks.sh
x64sc -reu -reusize 16384 -autostart examples/zeropage-reu.d64

Expect REAL LOAD OK and a green border. BOOT asks the KERNAL to load ENEMYRT into staging. ZeroPage caches it in the REU, copies it into the execution window, applies relocations, and dispatches through the export's jump-table slot. The optional result exposes a failed call to the program.

For several DLLs, see the typed multi-window example. Typed extern module bindings declare signatures in source. Load policies and reference counting govern when modules become callable. Configured state ranges can be saved and restored across activations. The same REU can also hold far arenas and typed data views accessed through explicit transfers. REU and DLL reference

Inspect and test the result

zp64c compile emits a PRG plus .asm, .labels, and .zpmap.json sidecars. Use the assembly to see the generated 6510 instructions, VICE labels to set breakpoints, and the map to inspect section ownership. zp64c asm emits assembly without running 64tass.

Terminal
zig build test
zig build emu-test
zig build e2e

The unit suite checks the compiler and runner. The Peddle host suite executes the generated machine code and includes test6502 assertions.
